# Elf Male - create and add to court
rsfantasy_summon.0011 = {
	type = character_event
	hidden = yes

	trigger = { exists = root.capital_province }

	immediate = {
		create_character = {
			template = rsfantasy_elf_summon_template
			location = root.capital_province
			culture = culture:rsfantasy_elf
			faith = faith:rsfantasy_elf_faith
			gender_female_chance = 0
			save_scope_as = summoned_courtier
		}
		if = {
			limit = { exists = scope:summoned_courtier }
			scope:summoned_courtier = {
				rsfantasy_set_spawn_age_16_28 = yes
				add_trait = rsfantasy_elf
				rsfantasy_grant_immortal_blessing_if_longlived_adult_effect = yes
			}
			root = {
				add_courtier = scope:summoned_courtier
				trigger_event = {
					id = rsfantasy_summon.0100
					days = 1
				}
			}
		}
	}
}

		if = {
			limit = { exists = scope:summoned_courtier }

				rsfantasy_set_spawn_age_16_28 = yes

				add_trait = rsfantasy_summoned_warrior
				rsfantasy_grant_immortal_blessing_if_longlived_adult_effect = yes
			}
			root = {
				add_courtier = scope:summoned_courtier
				trigger_event = { id = rsfantasy_summon.0100 days = 1 }
			}
		}
	}
}
