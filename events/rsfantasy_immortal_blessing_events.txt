namespace = rsfantasy_immortal_blessing

# One shot after lobby: any adult elf / dark elf / dwarf missing the modifier gets it (no per-character event spam).
rsfantasy_immortal_blessing.0001 = {
	scope = none
	hidden = yes

	immediate = {
		every_living_character = {
			if = {
				limit = {
					age >= 18
					OR = {
						has_trait = rsfantasy_elf
						has_trait = rsfantasy_dwarf
						has_trait = rsfantasy_dark_elf
					}
					NOT = { has_character_modifier = rsfantasy_immortal_blessing }
				}
				rsfantasy_grant_immortal_blessing_effect = yes
			}
		}
	}
}
