# SFantasy Mod - Fantasy race governments
# All adoptable via decisions only (AI will never form)

# Orcish Warband - Martial, prestige from battles, elective by prowess (feudal elective)
orcish_warband_government = {
	government_rules = {
		create_cadet_branches = yes
		rulers_should_have_dynasty = yes
		dynasty_named_realms = yes
		legitimacy = yes
	}

	royal_court = any
	fallback = 0

	primary_holding = castle_holding
	valid_holdings = { temple_citadel_holding }
	required_county_holdings = { castle_holding city_holding church_holding }

	vassal_contract_group = feudal_vassal

	can_get_government = {
		OR = {
			has_trait = rsfantasy_orc
			has_trait = rsfantasy_uruk_hai
		}
	}

	character_modifier = {
		martial = 2
		prowess = 1
		monthly_prestige = 0.25
		knight_effectiveness = 0.15
		army_maintenance_mult = -0.1
	}

	ai = { use_legends = yes }

	flags = {
		government_is_feudal
		government_is_settled
		may_elevate_co_monarch
		government_uses_crown_authority
		government_uses_domain_limit
		government_is_orcish_warband
	}

	mechanic_type = feudal
	color = hsv{ 0.15 0.8 0.5 }
	realm_mask_offset = { 0.0 0.01 }
	realm_mask_scale = { 1 1 }
}
