# SFantasy Mod - Summon fantasy race characters to court
# Player-only decisions. Cost: gold and prestige. Opens event to choose male/female.

summon_elf_to_court_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = minor

	desc = summon_elf_to_court_decision_desc

	ai_check_interval = 0

	is_shown = {
		is_ai = no
		exists = capital_province
	}

	is_valid = {
		is_ai = no
		gold >= 25
		prestige >= 25
	}

	cost = {
		gold = 25
		prestige = 25
	}

	effect = {
		trigger_event = rsfantasy_summon.0001
	}

	ai_will_do = { base = 0 }
}
